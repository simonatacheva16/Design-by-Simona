    <script>
        // ДАННИ ЗА ШАБЛОНИТЕ
        const templates = [
            {n: "Luxury Presentation", p: 15, i: "fa-chalkboard", limit: false, type: "presentation", style: "elegant", color: "gold", date: "2023-10-01", desc: "For real estate agents.", img: "url_to_image_1"},
            {n: "Bold Instagram Post", p: 5, i: "fa-instagram", limit: false, type: "social", style: "bold", color: "bordox", date: "2023-10-05", desc: "High engagement.", img: "url_to_image_2"},
            {n: "Minimalist Logo", p: 30, i: "fa-signature", limit: true, type: "business", style: "minimal", color: "black", date: "2023-10-10", desc: "Unique for one client.", img: "url_to_image_3"}
            // Добави тук всички 17+ шаблона!
        ];

        // Функция за генериране
        function renderTemplates(data) {
            const grid = document.getElementById('templateGrid');
            grid.innerHTML = '';
            
            data.forEach((item, index) => {
                grid.innerHTML += `
                    <div class="card" data-type="${item.type}" data-style="${item.style}" data-color="${item.color}">
                        ${item.limit ? '<div class="limit-badge">LIMITED</div>' : ''}
                        
                        <div class="card-img" onclick="openGallery('${item.img}')">
                            <i class="fas ${item.i} fa-2x"></i>
                            <span style="position:absolute; bottom:10px;">Click to view all slides</span>
                        </div>
                        
                        <h3>${item.n}</h3>
                        <span class="price">€${item.p}</span>
                        
                        <button class="details-btn" onclick="toggleDetails(${index})">View Details ▼</button>
                        <div class="card-details" id="details-${index}">
                            <p><strong>Style:</strong> ${item.style}</p>
                            <p><strong>Color:</strong> ${item.color}</p>
                            <p><strong>Created:</strong> ${item.date}</p>
                            <p><strong>Description:</strong> ${item.desc}</p>
                        </div>
                        
                        <button class="btn-add" onclick="addToCart('${item.n}', ${item.p})">Add to Package</button>
                    </div>
                `;
            });
        }

        function toggleDetails(index) {
            const details = document.getElementById(details-${index});
            details.style.display = details.style.display === 'block' ? 'none' : 'block';
        }

        function openGallery(imgUrl) {
            alert('Тук ще се отваря галерия с всички слайдове (Lightbox Gallery). Засега: ' + imgUrl);
        }

        // Филтриране
        function filterTemplates() {
            const typeValue = document.getElementById('typeFilter').value;
            const styleValue = document.getElementById('styleFilter').value;
            const colorValue = document.getElementById('colorFilter').value;
            
            const filtered = templates.filter(item => {
                return (typeValue === 'all' || item.type === typeValue) &&
                       (styleValue === 'all' || item.style === styleValue) &&
                       (colorValue === 'all' || item.color === colorValue);
            });
            
            renderTemplates(filtered);
        }

        // --- CART FUNCTIONALITY ---
        let cart = [];
        function showPage(pageId) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active-page'));
            document.getElementById(pageId).classList.add('active-page');
            window.scrollTo(0,0);
        }
        function addToCart(name, price) {
            cart.push({name, price});
            document.getElementById('cartCount').innerText = cart.length;
            updateCartUI();
        }
        function updateCartUI() {
            const display = document.getElementById('cartDisplay');
            const totalDisplay = document.getElementById('cartTotal');
            if(cart.length === 0) { display.innerHTML = "<p style='text-align:center;'>Package is empty.</p>"; totalDisplay.innerText = ""; } 
            else {
                display.innerHTML = cart.map((item, index) => `
                    <div class="cart-item">
                        <span>${item.name}</span>
                        <span>€${item.price} <i class="fas fa-times" onclick="removeItem(${index})" style="margin-left:15px; cursor:pointer; color:red;"></i></span>
                    </div>
                `).join('');
                const total = cart.reduce((sum, item) => sum + item.price, 0);
                totalDisplay.innerText = Total: €${total};
            }
        }
        function removeItem(index) {
            cart.splice(index, 1);
            document.getElementById('cartCount').innerText = cart.length;
            updateCartUI();
        }

        renderTemplates(templates);
    </script>
